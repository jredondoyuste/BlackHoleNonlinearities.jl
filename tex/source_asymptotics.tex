\pgfmathdeclarefunction{slogc}{2}{%
  \pgfmathparse{%
    (abs(#1) <= #2) ? (#1/#2) : ( ((#1<0) ? -1 : 1) * (1 + ln(abs(#1/#2))/ln(10)) )
  }%
}
